
local Lighting = game:GetService("Lighting")

local Enabled = false
local SavedParents = {}
local Connection = nil

local function ProcessEffect(effect)
    if not effect:IsA("Atmosphere") then
        return
    end

    if SavedParents[effect] == nil then
        SavedParents[effect] = effect.Parent
    end

    effect.Parent = nil
end

local function SetNoAtmosphere(state)
    Enabled = state == true

    if Enabled then
        for _, effect in ipairs(Lighting:GetDescendants()) do
            ProcessEffect(effect)
        end

        if not Connection then
            Connection = Lighting.DescendantAdded:Connect(function(effect)
                if Enabled then
                    ProcessEffect(effect)
                end
            end)
        end
    else
        if Connection then
            Connection:Disconnect()
            Connection = nil
        end

        for effect, originalParent in pairs(SavedParents) do
            if effect and originalParent then
                effect.Parent = originalParent
            end

            SavedParents[effect] = nil
        end
    end
end

return SetNoAtmosphere
